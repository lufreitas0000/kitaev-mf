def build_matter_A(L: int, K: float = 1.0, flipped_bonds: list[tuple[int, int, int]] | None = None) -> np.ndarray:
    """Assemble exact 2*L^2 x 2*L^2 real skew-symmetric matter Majorana generator A(u)."""
    N_cells = L * L
    N_sites = 2 * N_cells
    A = np.zeros((N_sites, N_sites), dtype=np.float64)
    flipped = set(flipped_bonds) if flipped_bonds else set()
    disp = [(0, 0), (0, -1), (-1, 0)]
    for m2 in range(L):
        for m1 in range(L):
            sA = 2 * (m1 + L * m2)
            for b_idx, (d1, d2) in enumerate(disp):
                n1 = (m1 + d1) % L
                n2 = (m2 + d2) % L
                sB = 2 * (n1 + L * n2) + 1
                sign = -1.0 if (m1, m2, b_idx) in flipped else 1.0
                A[sA, sB] += -2.0 * K * sign
                A[sB, sA] -= -2.0 * K * sign
    return A


def make_LR(w: np.ndarray, u: np.ndarray, tau: float) -> tuple[np.ndarray, np.ndarray]:
    """Bounded left/right decomposition e^{tau W} = L(tau)^{-1} R(tau) with entries in [0, 1]."""
    exp_neg = np.exp(-tau * np.abs(w))
    r_vals = np.where(w > 0, 1.0, exp_neg)
    l_vals = np.where(w > 0, exp_neg, 1.0)
    R = (u * r_vals) @ u.conj().T
    L = (u * l_vals) @ u.conj().T
    return L, R


def pfaffian_skew(M: np.ndarray) -> complex:
    """Exact Pfaffian of 2x2, 4x4, or 6x6 skew-symmetric matrix."""
    n = M.shape[0]
    if n == 2:
        return complex(M[0, 1])
    if n == 4:
        return complex(M[0, 1] * M[2, 3] - M[0, 2] * M[1, 3] + M[0, 3] * M[1, 2])
    if n == 6:
        res = 0.0j
        for j in range(1, 6):
            idx = [k for k in range(1, 6) if k != j]
            sub = M[np.ix_(idx, idx)]
            pf4 = sub[0, 1] * sub[2, 3] - sub[0, 2] * sub[1, 3] + sub[0, 3] * sub[1, 2]
            res += ((-1) ** (j - 1)) * M[0, j] * pf4
        return complex(res)
    raise ValueError(f"Unsupported Pfaffian dimension: {n}")


def wick_general(ops: list[tuple[int, int]], G_dict: dict[tuple[int, int], np.ndarray]) -> complex:
    """Multi-time Wick contraction via Pfaffian of time-slice Green blocks."""
    m = len(ops)
    M = np.zeros((m, m), dtype=complex)
    for p in range(m):
        sp, ip = ops[p]
        for q in range(p + 1, m):
            sq, iq = ops[q]
            M[p, q] = G_dict[(sp, sq)][ip, iq]
            M[q, p] = -M[p, q]
    return pfaffian_skew(M)


def eval_comm_general(mid_ops: list[tuple[int, int]], G_dict: dict[tuple[int, int], np.ndarray], a: int, b: int) -> complex:
    """Graded commutator <gs0 | {c_b, [c_a, O]} | gs0> / (4 Z_E) isolating normal-ordered bilinear."""
    return (
        + wick_general([(0, b), (0, a)] + mid_ops, G_dict)
        - wick_general([(0, b)] + mid_ops + [(2, a)], G_dict)
        + wick_general([(0, a)] + mid_ops + [(2, b)], G_dict)
        - wick_general(mid_ops + [(2, a), (2, b)], G_dict)
    ) / 4.0


def compute_torus_perturbation(L: int, n_quad_2nd: int = 24, n_quad_3rd: int = 10) -> dict[str, float]:
    """Compute exact 2nd-order and 3rd-order (Config a and Config b tadpole) coefficients on L x L PBC torus."""
    b_z, b_x, b_y = (0, 0, 0), (0, 0, 1), (0, 0, 2)
    s_l = 0
    s_jz = 1
    s_jx = 2 * (0 + L * (L - 1)) + 1
    s_ky = 2 * ((L - 1) + L * 0) + 1

    A0 = build_matter_A(L)
    Ax = build_matter_A(L, flipped_bonds=[b_x])
    Ay = build_matter_A(L, flipped_bonds=[b_y])
    Az = build_matter_A(L, flipped_bonds=[b_z])

    w0, u0 = np.linalg.eigh(1j * A0)
    wx, ux = np.linalg.eigh(1j * Ax)
    wy, uy = np.linalg.eigh(1j * Ay)
    wz, uz = np.linalg.eigh(1j * Az)
    E0 = -0.5 * float(np.sum(w0[w0 > 0]))
    dv = -0.5 * float(np.sum(wx[wx > 0])) - E0

    P_cre = u0[:, w0 < 0] @ u0[:, w0 < 0].conj().T
    P_ann = P_cre.T
    S0_bond = float(np.real(1j * (2.0 * P_ann[s_l, s_jz])))
    N = len(w0)
    Z_zero = np.zeros((N, N), dtype=complex)
    rhs_0 = np.vstack([2.0 * P_ann, Z_zero, Z_zero])

    # 1. Exact 2nd-order integral on bond b_z = (s_l, s_jz)
    P_cre_z = uz.conj().T @ P_cre @ uz
    P_ann_z = uz.conj().T @ P_ann @ uz
    uz_01 = uz[[s_l, s_jz], :]
    rhs_01 = P_ann_z @ uz_01.conj().T

    def integrand_2nd(tau: float) -> np.ndarray:
        exp_neg = np.exp(-tau * np.abs(wz))
        r_vals = np.where(wz > 0, 1.0, exp_neg)
        l_vals = np.where(wz > 0, exp_neg, 1.0)
        M_1 = P_cre_z * l_vals[None, :] + P_ann_z * r_vals[None, :]
        _, logdet = np.linalg.slogdet(M_1)
        Z_tau = float(np.exp(-tau * dv + 0.5 * np.real(logdet)))
        sol = np.linalg.solve(M_1, rhs_01)
        C_01 = 2.0 * (uz_01 * l_vals[None, :]) @ sol
        c00 = float(np.real(C_01[0, 0]))
        ic01 = float(np.real(1j * C_01[0, 1]))
        return np.array([-Z_tau * c00, -Z_tau * ic01, Z_tau])

    nodes2, weights2 = np.polynomial.legendre.leggauss(n_quad_2nd)
    t2 = 0.5 * (nodes2 + 1.0)
    tau2_pts = t2 / (1.0 - t2)
    jac2 = 0.5 / ((1.0 - t2) ** 2)
    acc2 = np.zeros(3)
    for tau, wg, jg in zip(tau2_pts, weights2, jac2):
        acc2 += wg * jg * integrand_2nd(float(tau))
    I_00, I_01, I_Z = float(acc2[0]), float(acc2[1]), float(acc2[2])

    # 2. Exact 3rd-order double integral (Config a Class I, Class II+III, and Config b star tadpole)
    def solve_pair_all6(wA: np.ndarray, uA: np.ndarray, tau1: float, wB: np.ndarray, uB: np.ndarray, tau2: float):
        LA, RA = make_LR(wA, uA, tau1)
        LB, RB = make_LR(wB, uB, tau2)
        M_AB = np.block([
            [P_cre, Z_zero, P_ann],
            [-RB,   LB,     Z_zero],
            [Z_zero, -RA,   LA]
        ])
        M_BA = np.block([
            [P_cre, Z_zero, P_ann],
            [-RA,   LA,     Z_zero],
            [Z_zero, -RB,   LB]
        ])
        sign, logdet = np.linalg.slogdet(M_AB)
        Z_val = np.exp(-(tau1 + tau2) * dv) * np.sqrt(sign * np.exp(logdet))
        rhs_AB = np.hstack([rhs_0, np.vstack([Z_zero, Z_zero, -2.0 * RA])])
        sol_AB = np.linalg.solve(M_AB, rhs_AB)
        sol_BA = np.linalg.solve(M_BA, rhs_0)
        return Z_val, {
            (0, 0): sol_BA[2 * N :, :].conj().T,
            (0, 1): sol_AB[:N, N:],
            (0, 2): sol_AB[:N, :N],
            (1, 1): sol_AB[N : 2 * N, N:],
            (1, 2): sol_AB[N : 2 * N, :N],
            (2, 2): sol_AB[2 * N :, :N],
        }

    sectors = {
        "x": (wx, ux, s_jx),
        "y": (wy, uy, s_ky),
        "z": (wz, uz, s_jz),
    }
    perms = [
        ("x", "y", "z", +1), ("y", "z", "x", +1), ("z", "x", "y", +1),
        ("x", "z", "y", -1), ("z", "y", "x", -1), ("y", "x", "z", -1),
    ]

    def integrand_3rd(tau1: float, tau2: float) -> np.ndarray:
        Z_xy, G_xy = solve_pair_all6(wy, uy, tau1, wx, ux, tau2)
        val_I = 2.0 * float(np.real(Z_xy * eval_comm_general([(0, s_jx), (2, s_ky)], G_xy, a=s_jx, b=s_ky)))
        Z_xz, G_xz = solve_pair_all6(wz, uz, tau1, wx, ux, tau2)
        val_II = 2.0 * float(np.real(
            Z_xz * eval_comm_general([(0, s_jx), (1, s_ky), (1, s_l), (2, s_l)], G_xz, a=s_jx, b=s_ky)
        ))
        tot_star = 0.0
        for kA, kB, kC, eps_s in perms:
            wA, uA, sA = sectors[kA]
            _, _, sB = sectors[kB]
            wC, uC, sC = sectors[kC]
            if kC == "z" and kA == "x":
                Z_CA, G_CA = Z_xz, G_xz
            elif kC == "y" and kA == "x":
                Z_CA, G_CA = Z_xy, G_xy
            else:
                Z_CA, G_CA = solve_pair_all6(wC, uC, tau1, wA, uA, tau2)
            c_val = eval_comm_general([(0, sA), (1, s_l), (1, sB), (2, sC)], G_CA, a=s_jx, b=s_ky)
            tot_star += eps_s * float(np.imag(Z_CA * c_val))
        return np.array([val_I, val_II, val_I + 2.0 * val_II, tot_star])

    nodes3, weights3 = np.polynomial.legendre.leggauss(n_quad_3rd)
    t3 = 0.5 * (nodes3 + 1.0)
    tau3_pts = t3 / (1.0 - t3)
    jac3 = 0.5 / ((1.0 - t3) ** 2)
    acc3 = np.zeros(4)
    for tau1, w1g, j1g in zip(tau3_pts, weights3, jac3):
        for tau2, w2g, j2g in zip(tau3_pts, weights3, jac3):
            acc3 += (w1g * j1g) * (w2g * j2g) * integrand_3rd(float(tau1), float(tau2))

    kappa_I, kappa_II, kappa_a, kappa_b_tadpole = (
        float(acc3[0]), float(acc3[1]), float(acc3[2]), float(acc3[3])
    )
    return {
        "L": L,
        "Delta_v": dv,
        "S0_bond": S0_bond,
        "I_00": I_00,
        "I_01": I_01,
        "I_Z": I_Z,
        "E2_bond_coeff": 2.0 * (I_00 + I_01),
        "eta_2_bond_renorm": -2.0 * I_01,
        "eff_inv_gap_2nd": -I_01 / S0_bond,
        "kappa_I": kappa_I,
        "kappa_II": kappa_II,
        "kappa_a_2f": kappa_a,
        "kappa_b_tadpole": kappa_b_tadpole,
        "kappa_eff_total": kappa_a + kappa_b_tadpole,
        "kappa_static_1_over_dv2": 1.0 / (dv ** 2),
    }
