\section{Extended Observables Suite and Definitions}
\label{sec:phase4_obs_definitions}
% CLAIM: K-026 (provenance: doc/rnd/phase_4/track_d_screening_dynamics_and_observables_survey.md, benchmark: data/processed/benchmarks/phase_4/four_theory_observables_comparison.dat)

To move beyond single-gap comparisons, we developed an extended suite of 10+ observables in \texttt{comprehensive\_observables.py}, evaluated across four theoretical frameworks on the $3 \times 3$ honeycomb torus ($N=18$ spins):
\begin{enumerate}
    \item \textbf{Ground Energy Density ($E_0 / N$):} Intensive ground-state energy.
    \item \textbf{Neutral Excitation Gap ($\Delta E$):} $\Delta E = E_1 - E_0$ in ED, compared with $\Delta_{\mathrm{canonical}}$ in MF and $\Delta_{\mathrm{Kit}}$ in perturbation theory.
    \item \textbf{Longitudinal Magnetization ($m_{111}$):} $m_{111} = \frac{\sqrt{3}}{N} \sum_i \langle \sigma_i^z \rangle$.
    \item \textbf{Differential Magnetic Susceptibility ($\chi_m$):} $\chi_m = \frac{\partial m_{111}}{\partial h} \approx \frac{m_{111}(h) - m_{111}(h - \Delta h)}{\Delta h}$.
    \item \textbf{Nearest-Neighbor Parallel Correlator ($C_{nn}^\parallel$):} On a $z$-bond, $C_{nn}^\parallel = \langle \sigma_i^z \sigma_j^z \rangle$.
    \item \textbf{Nearest-Neighbor Transverse Correlator ($C_{nn}^\perp$):} On a $z$-bond, $C_{nn}^\perp = \langle \sigma_i^x \sigma_j^x \rangle$.
    \item \textbf{Three-Spin Chiral Scalar Operator ($C_\chi$):} For three mutually connected sites meeting at a vertex, $C_\chi = \langle \sigma_i^x \sigma_j^y \sigma_k^z \rangle$.
    \item \textbf{Hexagonal Plaquette Flux ($\langle W_p \rangle$):} Expectation value of the hexagonal loop operator $W_p = \sigma_1^x \sigma_2^y \sigma_3^z \sigma_4^x \sigma_5^y \sigma_6^z$.
    \item \textbf{Bipartite Entanglement Entropy ($S_{\mathrm{vN}}$):} Von Neumann entropy across a spatial half-system cut dividing the $N=18$ torus into two 9-spin subsystems $A$ and $B$:
    \begin{equation}
    S_{\mathrm{vN}} = -\Tr(\rho_A \ln \rho_A) = -\sum_k s_k^2 \ln(s_k^2),
    \end{equation}
    where $s_k$ are the Schmidt singular values of the reshaped state matrix $\Psi_{a, b}$.
    \item \textbf{Fidelity Susceptibility ($\chi_F$):} $\chi_F = \lim_{\Delta h \to 0} \frac{2(1 - |\langle \Psi(h) | \Psi(h - \Delta h) \rangle|)}{(\Delta h)^2}$.
    \item \textbf{Fixed-Point Contraction Factor ($\mu$) and Relaxation Rate ($\Gamma$):} $\mu = \lim_{k \to \infty} \frac{\|\mathbf{P}^{(k+1)} - \mathbf{P}^{(k)}\|}{\|\mathbf{P}^{(k)} - \mathbf{P}^{(k-1)}\|}$, $\Gamma = -\ln(\mu)$.
\end{enumerate}

\subsection{Physical Inferences}
\begin{enumerate}
    \item \textbf{Thermodynamic Convergence:} The ED ground energy on $N=18$ ($E_0/N = -0.793972$) agrees with the infinite-lattice mean-field energy ($-0.787299$) within $0.85\%$, confirming negligible finite-size corrections.
    \item \textbf{Quantum Entanglement:} The bipartite entanglement entropy across the half-torus cut ($9 \times 9$ spins) is $S_{\mathrm{vN}} = 3.2289 \approx 4.66 \ln(2)$, verifying extensive entanglement characteristic of a deconfined gauge theory. As $h$ increases, $S_{\mathrm{vN}}$ decreases monotonically to $2.8567$ at $h=0.05$ as the spins polarize.
    \item \textbf{Scalar Chiral Order ($C_\chi$):} The three-spin scalar chiral operator grows linearly with $h$ at low fields ($C_\chi \approx 0.68 h$), directly measuring the microscopic breaking of time-reversal symmetry that opens the bulk topological Chern gap.
\end{enumerate}
