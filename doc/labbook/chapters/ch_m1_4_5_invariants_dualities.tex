\section{Kinematic and Semidefinite Representability Gates}
In non-integrable parameter regimes, exact ground states are unknown. Verification relies on universal kinematic, semidefinite, and variational invariants:
\begin{itemize}
    \item \textbf{Parseval-Frobenius Identity}: On any regular grid ($N_1, N_2 \ge 2$), $\frac{1}{N_{\mathbf{k}}}\sum \mathcal{E}_n^2 \equiv \frac{1}{16}\sum \Vert{}B_s\Vert{}_{\mathrm{F}}^2 + \frac{1}{8}\sum \Vert{}A_b\Vert{}_{\mathrm{F}}^2$.
    \item \textbf{Local Two-Site LMI Positivity}: The $8 \times 8$ matrix $\mathcal{C}_{ij} = \langle \mathbf{c}_{ij} \mathbf{c}_{ij}^T \rangle$ satisfies $\lambda_{\min}(\mathcal{C}_{ij}) \ge 0$.
    \item \textbf{Operator Norm Bound}: Inter-site hoppings satisfy $\sigma_{\max}(\mathrm{U}_b) \le 1.0$.
    \item \textbf{Bloch-Ball \& Pfaffian Parity Envelopes}: $\Vert{}\langle \boldsymbol{\sigma}_s \rangle\Vert{}_2 \le 1.0$ and $-1.0 \le \operatorname{Pf}(\mathrm{V}_s) \le 1.0$.
\end{itemize}

\section{Solvable Reductions and Discrete Symmetries}
\begin{itemize}
    \item \textbf{1D Zigzag Chain Limit} ($K_x = K_y = -1, K_z = 0$): Active bond hoppings evaluate to $U_x^{00} = U_y^{00} = -2/\pi \approx -0.636620$, while $U_z \equiv \mathbf{0}$.
    \item \textbf{Isolated Dimer Flat Bands} ($K_z = -1, K_x = K_y = 0$): Bands form flat levels at $\mathcal{E} = \pm 0.5$ and $\mathcal{E} = 0$.
    \item \textbf{Discrete Selection Rules}: Time-reversal-odd correlators vanish ($U_b^{0\gamma} \equiv 0$), sublattice inversion parity holds ($V_A = V_B$), and $C_3$ ray permutation invariance is preserved.
\end{itemize}
