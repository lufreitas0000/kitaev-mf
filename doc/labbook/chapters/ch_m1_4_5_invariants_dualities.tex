\section{Kinematic Invariants and Semidefinite Representability Gates}
\label{sec:m1_4_5_kinematic_gates}
% CLAIM: CLM-M1-KIN-001 (provenance: doc/rnd/phase_1/m1_4_5_t1_kinematic_and_semidefinite_invariants.md, test: test/unit/test_kinematic_and_semidefinite.py::test_kinematic_t01_parseval_frobenius_spectral_sum_rule)
In non-integrable parameter regimes ($J, \Gamma, \Gamma', \mathbf{h} \ne 0$), exact analytical solutions are unavailable. Rigorous verification of the mean-field solution requires universal kinematic conservation laws and semidefinite programming (SDP) representability gates that hold identically across all couplings:

\begin{theorem}[Parseval-Frobenius Spectral Energy Sum Rule]
\label{thm:m1_4_5_parseval_frobenius}
On any regular Brillouin-zone grid of $N_{\mathbf{k}}$ momenta with $N_1, N_2 \ge 2$, the sum of squared Bloch eigenvalues is exactly conserved by the real-space matrix Frobenius norms:
\begin{equation}
\label{eq:m1_4_5_parseval_sum_rule}
\frac{1}{N_{\mathbf{k}}} \sum_{\mathbf{k} \in \mathrm{BZ}} \sum_{n=1}^{4N_{\mathrm{sub}}} \mathcal{E}_n^2(\mathbf{k}) = \frac{1}{16} \sum_{s=1}^{N_{\mathrm{sub}}} \|\mathrm{B}_s\|_{\mathrm{F}}^2 + \frac{1}{8} \sum_{b=1}^{N_{\mathrm{bonds}}} \|\mathrm{A}_b\|_{\mathrm{F}}^2.
\end{equation}
\end{theorem}
\begin{proof}
Because $\mathcal{H}(\mathbf{k}) = \frac{\ii}{4}\mathcal{M}(\mathbf{k})$ is Hermitian, $\sum_n \mathcal{E}_n^2(\mathbf{k}) = \Tr(\mathcal{H}(\mathbf{k})^2) = \frac{1}{16} \Tr(\mathcal{M}(\mathbf{k})^\dagger \mathcal{M}(\mathbf{k}))$. Expanding in real-space block matrices:
\begin{equation}
\Tr(\mathcal{M}(\mathbf{k})^\dagger \mathcal{M}(\mathbf{k})) = \sum_{s=1}^{N_{\mathrm{sub}}} \|\mathrm{B}_s\|_{\mathrm{F}}^2 + \sum_{b=1}^{N_{\mathrm{bonds}}} \|\mathrm{A}_b\|_{\mathrm{F}}^2 + \sum_{b \ne b'} \Tr(\mathrm{A}_b^T \mathrm{A}_{b'}) e^{\ii (\phi_b - \phi_{b'})}.
\end{equation}
Averaging over any regular mesh $N_1, N_2 \ge 2$, the cross-phase interference terms average to zero by discrete orthogonality: $\frac{1}{N_{\mathbf{k}}} \sum_{\mathbf{k}} e^{\ii (\phi_b - \phi_{b'})} = \delta_{bb'}$. Noting that directed bonds connect pairs of sublattices, the off-diagonal contribution carries an extra factor of $2$, yielding Eq.~\eqref{eq:m1_4_5_parseval_sum_rule}.
\end{proof}

\begin{theorem}[Two-Site Semidefinite Covariance Positivity (LMI)]
\label{thm:m1_4_5_two_site_lmi}
Let $\mathbf{c}_{ij} = (c_i^0, \dots, c_i^3, c_j^0, \dots, c_j^3)^T$ be the 8-Majorana basis on bond $\langle ij \rangle$. The local two-site correlation matrix $\mathcal{C}_{ij} \equiv \langle \mathbf{c}_{ij} \mathbf{c}_{ij}^T \rangle_{\mathrm{G}} \in \mathbb{C}^{8 \times 8}$ expands as:
\begin{equation}
\label{eq:m1_4_5_two_site_C_matrix}
\mathcal{C}_{ij} = \begin{pmatrix} \Id_4 - \ii \mathrm{V}_i & -\ii \mathrm{U}_{ij} \\ \ii \mathrm{U}_{ij}^T & \Id_4 - \ii \mathrm{V}_j \end{pmatrix}.
\end{equation}
Quantum representability requires that $\mathcal{C}_{ij}$ be positive semidefinite:
\begin{equation}
\label{eq:m1_4_5_lmi_condition}
\mathcal{C}_{ij} \succeq 0 \iff \lambda_{\min}(\mathcal{C}_{ij}) \ge 0.
\end{equation}
When on-site magnetic and gauge polarizations vanish ($\mathrm{V}_i = \mathrm{V}_j = \mathbf{0}$), Eq.~\eqref{eq:m1_4_5_lmi_condition} is equivalent to the operator norm contraction bound:
\begin{equation}
\label{eq:m1_4_5_operator_norm_bound}
\|\mathrm{U}_{ij}\|_2 \le 1.0.
\end{equation}
\end{theorem}
\begin{proof}
For any complex test vector $\mathbf{z} \in \mathbb{C}^8$, the expectation value $\langle (\mathbf{z}^\dagger \mathbf{c}_{ij})(\mathbf{c}_{ij}^T \mathbf{z}) \rangle_{\mathrm{G}} = \mathbf{z}^\dagger \mathcal{C}_{ij} \mathbf{z} \ge 0$ by positivity of the density matrix. When $\mathrm{V} = \mathbf{0}$, $\mathcal{C}_{ij} = \Id_8 - \ii \begin{pmatrix} 0 & \mathrm{U} \\ -\mathrm{U}^T & 0 \end{pmatrix}$. The eigenvalues of $\ii \begin{pmatrix} 0 & \mathrm{U} \\ -\mathrm{U}^T & 0 \end{pmatrix}$ are $\pm \sigma_k(\mathrm{U})$. Thus $\lambda_{\min}(\mathcal{C}_{ij}) = 1 - \sigma_{\max}(\mathrm{U}) \ge 0 \iff \|\mathrm{U}\|_2 \le 1$.
\end{proof}

\begin{proposition}[Bloch-Ball and Pfaffian Parity Envelopes]
\label{prop:m1_4_5_parity_envelopes}
On each sublattice site $s$, the local spin magnetization vector $\mathbf{m}_s = (m_s^x, m_s^y, m_s^z)$ and the local on-site Pfaffian parity satisfy the strict containment bounds:
\begin{equation}
\label{eq:m1_4_5_bloch_pfaffian_envelopes}
\|\mathbf{m}_s\|_2 \le 1.0, \qquad -1.0 \le \operatorname{Pf}(\mathrm{V}_s) \le 1.0.
\end{equation}
\end{proposition}

\section{Solvable Reductions and Discrete Selection Rules}
\label{sec:m1_4_5_solvable_reductions}
% CLAIM: CLM-M1-RED-001 (provenance: doc/rnd/phase_1/m1_4_5_t1_kinematic_and_semidefinite_invariants.md, test: test/unit/test_solvable_limits.py::test_solvable_t01_1d_zigzag_chain_reduction)

\begin{theorem}[Analytical 1D Zigzag Chain Limit]
\label{thm:m1_4_5_1d_zigzag}
In the anisotropic limit where $z$-bonds are decoupled ($K_x = K_y = -1.0, K_z = 0$), the 2D honeycomb lattice decouples into independent 1D zigzag chains along the $x$-direction. The matter dispersion reduces to:
\begin{equation}
\label{eq:m1_4_5_zigzag_dispersion}
\mathcal{E}(k_x) = \frac{|K|}{2} |1 + e^{-\ii 2\pi k_x}| = |K| |\cos(\pi k_x)|.
\end{equation}
The self-consistent ground energy per site evaluates to the closed-form integral:
\begin{equation}
\label{eq:m1_4_5_zigzag_energy}
E_0^{\mathrm{1D}} = -\frac{|K|}{\pi} \int_{-\pi/2}^{\pi/2} \cos\theta \, \dd\theta = -\frac{2|K|}{\pi} \approx -0.6366197723676,
\end{equation}
with active matter hoppings $\mathrm{U}_x^{00} = \mathrm{U}_y^{00} = -\frac{2}{\pi}$ and strictly vanishing inter-chain hopping $\mathrm{U}_z \equiv \mathbf{0}$.
\end{theorem}

\begin{proposition}[Isolated Dimer Flat Bands]
\label{prop:m1_4_5_dimer_reduction}
In the complementary dimer limit where $x$- and $y$-bonds vanish ($K_z = -1.0, K_x = K_y = 0$), all dispersion ceases. The Bloch spectrum collapses into completely flat localized levels at $\mathcal{E} = \pm 0.5$ and $\mathcal{E} = 0$, with isolated $z$-bond hopping $\mathrm{U}_z^{00} = -1.0$.
\end{proposition}

\begin{proposition}[Discrete Symmetries and Selection Rules]
\label{prop:m1_4_5_selection_rules}
In the absence of external magnetic fields ($\mathbf{h} = \mathbf{0}$):
\begin{enumerate}
\item \textbf{Time-Reversal Parity:} Correlators between matter and gauge Majoranas vanish identically: $\mathrm{U}_b^{0\gamma} \equiv 0$ ($\gamma \in \{1, 2, 3\}$).
\item \textbf{Sublattice Inversion Symmetry:} On-site covariances satisfy $\mathrm{V}_A = \mathrm{V}_B$.
\item \textbf{Cubic Permutation Parity ($C_3$):} Under cyclic permutation of bond directions $(x, y, z)$, the corresponding bond tensors map into each other under $C_3$ symmetry.
\end{enumerate}
\end{proposition}

\section{Variational Response and Hellmann-Feynman Error Scaling}
\label{sec:m1_4_5_variational_response}

At self-consistent variational stationarity on the Grassmannian, the Hellmann-Feynman theorem dictates that the field derivative equals the physical total magnetization:
\begin{equation}
\label{eq:m1_4_5_hellmann_feynman_identity}
\frac{\dd E_{\mathrm{cell}}}{\dd h^\gamma} = -\langle \sigma_{\mathrm{tot}}^\gamma \rangle_{\mathrm{G}}.
\end{equation}
When computing numerical derivatives via central differences $(\Delta E / \Delta h)_{\mathrm{num}} = \frac{E(h + \delta) - E(h - \delta)}{2\delta}$, the total discrepancy decomposes as:
\begin{equation}
\label{eq:m1_4_5_central_diff_error}
\Delta_{\mathrm{HF}}(\delta) \equiv \left| \left(\frac{\Delta E}{\Delta h}\right)_{\mathrm{num}} - (-\langle \sigma_{\mathrm{tot}}^z \rangle) \right| \le \frac{1}{6} |E'''(h)| \delta^2 + C_\sigma \tau + C_E \frac{\tau^2}{\delta},
\end{equation}
where $\tau$ is the fixed-point solver residual tolerance.

\label{sec:m1_4_5_klein_duality}

 $\mathcal{K}$ performs a canonical $\pi$-rotation around alternating spin axes on four sublattices:
\begin{equation}
\label{eq:m1_4_5_klein_map}
\mathcal{K}: (J, K, \Gamma, \Gamma') \mapsto (-J, K + 2J, -\Gamma, \Gamma').
\end{equation}
On an infinite honeycomb lattice, $\mathcal{K}$ is an exact unitary equivalence. On a finite periodic cell of $N_{\mathrm{sub}} = 4$ sublattices, the boundary condition introduces a finite-size splitting $\delta E_{\mathrm{site}}(\boldsymbol{\epsilon}) \equiv E_{\mathrm{site}}(\mathbf{p}) - E_{\mathrm{site}}(\mathcal{K}(\mathbf{p}))$, where $\boldsymbol{\epsilon} = (\epsilon_J, \epsilon_\Gamma)$.

\begin{theorem}[Odd-Parity Klein Boundary Splitting Law]
\label{thm:m1_4_5_klein_cubic_law}
Under parameter inversion $\boldsymbol{\epsilon} \mapsto -\boldsymbol{\epsilon}$, the finite-cell energy splitting satisfies the odd-parity reflection symmetry:
\begin{equation}
\label{eq:m1_4_5_klein_odd_parity}
\delta E_{\mathrm{site}}(-\boldsymbol{\epsilon}) = -\delta E_{\mathrm{site}}(\boldsymbol{\epsilon}).
\end{equation}
Because the Taylor expansion of $\delta E_{\mathrm{site}}(\boldsymbol{\epsilon})$ about the unperturbed Kitaev point contains only odd powers, $c_0 = c_2 = 0$. Furthermore, because the unperturbed Kitaev ground state possesses exact Klein dual symmetry, the linear coefficient vanishes identically: $c_1 \equiv 0$.
Consequently, the finite-cell boundary splitting vanishes **strictly cubically**:
\begin{equation}
\label{eq:m1_4_5_klein_cubic_scaling}
\delta E_{\mathrm{site}}(\boldsymbol{\epsilon}) = c_3 \epsilon^3 + \mathcal{O}(\epsilon^5).
\end{equation}
\end{theorem}
\begin{proof}
The Klein involution $\mathcal{K}$ satisfies $\mathcal{K}^2 = \Id$. Near the pure Kitaev point $\mathbf{p}_0 = (0, K, 0, 0)$, $\mathcal{K}(\mathbf{p}_0 + \boldsymbol{\epsilon}) = \mathbf{p}_0 - \boldsymbol{\epsilon} + \mathcal{O}(\epsilon^2)$. Under reflection $\boldsymbol{\epsilon} \to -\boldsymbol{\epsilon}$, the roles of the direct and dual parameter configurations are interchanged, enforcing $\delta E(-\boldsymbol{\epsilon}) = -\delta E(\boldsymbol{\epsilon})$. Analyticity of the ground-state energy functional implies that all even Taylor coefficients vanish ($c_{2m} = 0$). Since the pure Kitaev point is dual-invariant ($\delta E(\mathbf{0}) = 0$), the first derivative $\nabla_{\boldsymbol{\epsilon}} \delta E(\mathbf{0}) = \mathbf{0}$ by symmetry, proving that $c_1 = 0$ and the leading non-vanishing term is $\mathcal{O}(\epsilon^3)$.
\end{proof}
